class AdaptiveFeatureModulation(nn.Module):
    def __init__(self, channels, condition_dim):
        super().__init__()
        self.fc = nn.Sequential(nn.Linear(condition_dim, channels // 2),
                                nn.ReLU(),
                                nn.Linear(channels // 2, channels * 2))
    def forward(self, x, scale_embedding):
        gamma_beta = self.fc(scale_embedding).view(-1, x.size(1), 1, 1)
        gamma, beta = gamma_beta.chunk(2, dim=1)
        return x * (1 + gamma) + beta

3) Multi-Scale Atrous Pyramid (MSAP) with dynamic dilations
class MultiScaleAtrousPyramid(nn.Module):
    def __init__(self, in_channels, out_channels, num_branches=4):
        super().__init__()
        dilations = [1, 2, 4, 8][:num_branches]
        self.branches = nn.ModuleList([
            nn.Conv2d(in_channels, out_channels, kernel_size=3, padding=d, dilation=d)
            for d in dilations
        ])
        self.fuse = nn.Conv2d(out_channels * num_branches, out_channels, 1)
        # optional per-channel gating learned factors can be added here
    def forward(self, x):
        feats = [b(x) for b in self.branches]
        cat = torch.cat(feats, dim=1)
        return self.fuse(cat)

class ScaleAwareFusionModule(nn.Module):
    def __init__(self, feats_channels, scale_dim):
        super().__init__()
        self.channel_att = nn.Sequential(nn.Conv2d(sum(feats_channels), sum(feats_channels) // 2, 1),
                                         nn.ReLU(),

        self.pos_att = nn.Sequential(nn.Conv2d(sum(feats_channels), sum(feats_channels) // 2, 1),

                                     nn.Softmax(dim=-1))
        self.uncertainty_head = nn.Conv2d(sum(feats_channels), 1, 1)
    def forward(self, feats_list, scale_embedding):

- Channel attention: compute channel logits Lc = Conv1x1(ReLU(Conv1x1(cat_feats))) with shape [B, Ctot, H, W]. Channel attention weights wc = softmax(global_avg_pool(Lc), dim=1) where global_avg_pool reduces (H,W) → [B, Ctot, 1, 1]. Softmax dim=1 (across channels). Apply: fused = cat_feats * wc.
